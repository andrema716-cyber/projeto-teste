function confirmCheckout() { if (!cart.length) { showToast('O carrinho está vazio!'); return; } // Confere o estoque antes de alterar qualquer dado for (const item of cart) { const produto = db.products.find( p => p.code === item.code ); if (!produto) { showToast(`Produto não encontrado: ${item.name}`); return; } if (!Number.isInteger(item.qty) || item.qty <= 0) { showToast('Quantidade inválida no carrinho!'); return; } if (produto.stock < item.qty) { showToast( `Estoque insuficiente para ${produto.name}. ` + `Disponível: ${produto.stock}` ); return; } } // Cálculos em centavos const totalCentavos = cart.reduce( (soma, item) => soma + paraCentavos(item.price) * item.qty, 0 ); const custoCentavos = cart.reduce( (soma, item) => soma + paraCentavos(item.cost) * item.qty, 0 ); const total = totalCentavos / 100; const custo = custoCentavos / 100; const lucro = (totalCentavos - custoCentavos) / 100; let cliente = null; // Validação do pagamento em dinheiro if (currentPaymentMethod === 'Dinheiro') { const recebido = Number( document.getElementById('cashReceivedInput').value ); const recebidoCentavos = Number.isFinite(recebido) ? Math.round(recebido * 100) : 0; if (recebidoCentavos < totalCentavos) { showToast('O valor recebido é insuficiente!'); return; } } // Validação do fiado if (currentPaymentMethod === 'Fiado') { const idx = document.getElementById( 'checkoutCustomerSelect' ).value; cliente = db.customers[Number(idx)]; if (idx === '' || !cliente) { showToast('Selecione um cliente válido!'); return; } const saldoAtual = paraCentavos(cliente.balance || 0); const limite = paraCentavos(cliente.limit); const novoSaldo = saldoAtual + totalCentavos; if (novoSaldo > limite) { showToast('Limite de crédito excedido!'); return; } } // Atualiza o estoque e a lista de reposição for (const item of cart) { const produto = db.products.find( p => p.code === item.code ); produto.stock -= item.qty; let reposicao = db.restock.find( r => r.code === produto.code ); if (reposicao) { reposicao.qty += item.qty; reposicao.name = produto.name; } else { db.restock.push({ code: produto.code, name: produto.name, qty: item.qty }); } } // Registra a dívida do cliente if (cliente) { cliente.balance = ( paraCentavos(cliente.balance || 0) + totalCentavos ) / 100; cliente.history = cliente.history || []; cliente.history.push({ date: new Date().toISOString(), type: 'compra_fiado', amount: total, balanceAfter: cliente.balance }); } // Só o dinheiro em espécie entra neste saldo if (currentPaymentMethod === 'Dinheiro') { const recebidoCentavos = Math.round( Number( document.getElementById('cashReceivedInput').value ) * 100 ); const trocoCentavos = recebidoCentavos - totalCentavos; db.cashDrawer = ( paraCentavos(db.cashDrawer) + totalCentavos ) / 100; // O troco entregue não permanece no caixa db.cashDrawer = ( paraCentavos(db.cashDrawer) - trocoCentavos ) / 100; } // Registra a venda db.sales.push({ id: Date.now(), date: new Date().toISOString(), items: cart.map(item => ({ ...item })), total, cost: custo, profit: lucro, method: currentPaymentMethod, customer: cliente ? cliente.name : null }); saveDB(); closePaymentModal(); showToast('Venda concluída com sucesso!'); cancelSale(); // Atualiza as telas que dependem dos dados renderProductList(); renderRestockList(); renderHistory(); renderCustomerList(); }
